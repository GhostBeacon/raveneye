# RavenEye - Konfiguration
# Datei auf die SD-Karte kopieren: /raveneye/config.txt
# Echte Werte nur auf der Karte, nie ins Repo.

# Mehrere WLANs moeglich: wifi_*, wifi2_*, ... wifi9_*.
# Verbunden wird mit dem staerksten, das gerade erreichbar ist.
wifi_ssid=<WLAN_NAME>
wifi_pass=<WLAN_PASSWORT>
wifi2_ssid=<WLAN2_NAME>
wifi2_pass=<WLAN2_PASSWORT>

# nur Hostname, Verbindung immer per HTTPS
ntfy_server=ntfy.sh
# mehrere Topics mit Komma, z. B. alarm,info
ntfy_topics=<NTFY_TOPIC>

# Statusanzeige (optional): oeffentliche Statusseite von Uptime Kuma 1.x
# Adresse ohne Pfad, dazu der Slug der Statusseite (steht in deren URL /status/<slug>)
status_url=https://<STATUS_HOST>
status_slug=<SLUG>

# Dateien (optional): Dateimanager mit JSON-API unter <cloud_url>/api/v1
# Anmeldung erfolgt auf dem Geraet, Passwoerter gehoeren NICHT in diese Datei.
cloud_url=https://<CLOUD_HOST>
